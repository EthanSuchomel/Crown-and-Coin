func _make_font(weight: int) -> Font:
	var oswald: String = _find_oswald("res://art/fonts")
	if oswald != "":
		var file: Font = load(oswald)
		var fv := FontVariation.new()
		fv.base_font = file
		if oswald.to_lower().contains("variable"):
			var ts: TextServer = TextServerManager.get_primary_interface()
			fv.variation_opentype = {ts.name_to_tag("wght"): weight}
		elif weight >= 600:
			fv.variation_embolden = 0.6   # a single static Oswald file: Godot makes the bold weight itself
		return fv
	var sf := SystemFont.new()
	sf.font_names = PackedStringArray(["Oswald", "Bahnschrift", "Arial Narrow", "Roboto Condensed", "Arial"])
	sf.font_weight = weight
	sf.multichannel_signed_distance_field = true   # stays sharp at any size
	sf.generate_mipmaps = true
	return sf


## Finds an Oswald .ttf anywhere inside the fonts folder (and its subfolders),
## preferring the variable font that holds every weight.
func _find_oswald(dir_path: String) -> String:
	var d := DirAccess.open(dir_path)
	if d == null:
		return ""
	var found: String = ""
	for f in d.get_files():
		var fname: String = f.trim_suffix(".import").trim_suffix(".remap")
		var low: String = fname.to_lower()
		if low.begins_with("oswald") and low.ends_with(".ttf"):
			var path: String = dir_path.path_join(fname)
			if low.contains("variable"):
				return path
			if found == "" or low.contains("regular"):
				found = path
	for sub in d.get_directories():
		var inner: String = _find_oswald(dir_path.path_join(sub))
		if inner.to_lower().contains("variable"):
			return inner
		if found == "" and inner != "":
			found = inner
	return found


func _box(fill: Color, border: Color, width: int = 2, margin: float = 8.0) -> StyleBoxFlat:
	var sb := StyleBoxFlat.new()
	sb.bg_color = fill
	sb.border_color = border
	sb.set_border_width_all(width)
	sb.set_corner_radius_all(3)
	sb.set_content_margin_all(margin)
	sb.shadow_color = OUTLINE
	sb.shadow_size = 2
	sb.anti_aliasing = false
	return sb


## The pixel frame used everywhere: a moss line inside a near-black outline.
func _frame(fill: Color, margin: float = 8.0) -> StyleBoxFlat:
	return _box(fill, EDGE, 2, margin)


func _button_box(fill: Color, bottom: Color) -> StyleBoxFlat:
	var sb := StyleBoxFlat.new()
	sb.bg_color = fill
	sb.border_color = OUTLINE
	sb.set_border_width_all(2)
	sb.border_width_bottom = 4
	sb.set_corner_radius_all(3)
	sb.content_margin_left = 12
	sb.content_margin_right = 12
	sb.content_margin_top = 7
	sb.content_margin_bottom = 8
	sb.anti_aliasing = false
	# a darker strip along the bottom, like the browser's inset shadow
	sb.expand_margin_bottom = 0
	sb.shadow_color = Color(bottom, 0.0)
	return sb


func _make_theme() -> Theme:
	var th := Theme.new()
	th.default_font = _font_body
	th.default_font_size = 16
	th.set_color("font_color", "Label", BONE)
	th.set_color("default_color", "RichTextLabel", BONE)
	th.set_font("bold_font", "RichTextLabel", _font_bold)
	th.set_font("normal_font", "RichTextLabel", _font_body)
	# buttons: moss with a near-black outline
	th.set_font("font", "Button", _font_bold)
	th.set_stylebox("normal", "Button", _button_box(MOSS2, OUTLINE))
	th.set_stylebox("hover", "Button", _button_box(Color("243a29"), OUTLINE))
	th.set_stylebox("pressed", "Button", _button_box(Color("142118"), OUTLINE))
	th.set_stylebox("disabled", "Button", _button_box(Color(MOSS2, 0.45), OUTLINE))
	var focus := StyleBoxFlat.new()
	focus.draw_center = false
	focus.border_color = GOLD
	focus.set_border_width_all(2)
	focus.set_corner_radius_all(3)
	th.set_stylebox("focus", "Button", focus)
	th.set_color("font_color", "Button", BONE)
	th.set_color("font_hover_color", "Button", BONE)
	th.set_color("font_pressed_color", "Button", BONE)
	th.set_color("font_focus_color", "Button", BONE)
	th.set_color("font_disabled_color", "Button", Color(BONE, 0.4))
	# the gold primary button
	th.set_type_variation("Primary", "Button")
	th.set_stylebox("normal", "Primary", _button_box(GOLD, GOLD2))
	th.set_stylebox("hover", "Primary", _button_box(Color("f6cd6c"), GOLD2))
	th.set_stylebox("pressed", "Primary", _button_box(GOLD2, GOLD2))
	th.set_stylebox("disabled", "Primary", _button_box(Color(GOLD, 0.4), GOLD2))
	for c in ["font_color", "font_hover_color", "font_pressed_color", "font_focus_color"]:
		th.set_color(c, "Primary", GOLD_INK)
	th.set_color("font_disabled_color", "Primary", Color(GOLD_INK, 0.6))
	# a card in your hand: a flat button whose content is drawn by child controls
	th.set_type_variation("CardButton", "Button")
	var card := _frame(MOSS2, 0)
	th.set_stylebox("normal", "CardButton", card)
	var card_hover := _frame(Color("243a29"), 0)
	th.set_stylebox("hover", "CardButton", card_hover)
	th.set_stylebox("pressed", "CardButton", card_hover)
	var card_due := _box(MOSS2, GOLD, 2, 0)
	th.set_type_variation("CardDue", "Button")
	th.set_stylebox("normal", "CardDue", card_due)
	th.set_stylebox("hover", "CardDue", _box(Color("243a29"), GOLD, 2, 0))
	th.set_stylebox("pressed", "CardDue", _box(Color("243a29"), GOLD, 2, 0))
	# panels
	th.set_stylebox("panel", "PanelContainer", _frame(MOSS))
	th.set_stylebox("panel", "PopupPanel", _frame(MOSS))
	# spin boxes and scroll bars
	th.set_color("font_color", "LineEdit", BONE)
	th.set_stylebox("normal", "LineEdit", _box(NIGHT, EDGE, 2, 4))
	th.set_stylebox("scroll", "VScrollBar", _box(NIGHT, NIGHT, 0, 2))
	th.set_stylebox("grabber", "VScrollBar", _box(EDGE, EDGE, 0, 2))
	return th

func _clear() -> void:
	for c in get_children():
		c.queue_free()
	overlay = null
	var bg := ColorRect.new()
	bg.color = NIGHT
	bg.set_anchors_preset(Control.PRESET_FULL_RECT)
	add_child(bg)


## The grass backdrop with a dark vignette, behind the title and faction screens.
func _backdrop() -> void:
	var tile := TextureRect.new()
	tile.texture = load("res://art/ui/title_bg.png")
	tile.stretch_mode = TextureRect.STRETCH_TILE
	tile.set_anchors_preset(Control.PRESET_FULL_RECT)
	add_child(tile)
	var grad := Gradient.new()
	grad.set_color(0, Color(0.035, 0.063, 0.043, 0.5))
	grad.set_color(1, Color(0.035, 0.063, 0.043, 0.92))
	var gt := GradientTexture2D.new()
	gt.gradient = grad
	gt.fill = GradientTexture2D.FILL_RADIAL
	gt.fill_from = Vector2(0.5, 0.5)
	gt.fill_to = Vector2(1.05, 1.05)
	var vignette := TextureRect.new()
	vignette.texture = gt
	vignette.stretch_mode = TextureRect.STRETCH_SCALE
	vignette.expand_mode = TextureRect.EXPAND_IGNORE_SIZE
	vignette.set_anchors_preset(Control.PRESET_FULL_RECT)
	add_child(vignette)

## Every Dinosaur or Robot card's own upgrades, for How to play and the faction screen.
func _card_list(fid: String) -> String:
	var lines: Array = ["[b]Every %s card has its own upgrades for each of its actions[/b] (every 2nd play you pick which action goes up a level):" % ("Robot" if fid == "robots" else "Dinosaur")]
	for c in GameData.factions[fid].cards:

		for a in c.actions:
			var d: Array = GameData.upgrades.get(fid + "_cards", {}).get(c.id, {}).get(a, [])
			var lv: Array = []
			for i in d.size():
				lv.append("%d %s: %s." % [i + 1, d[i][0], d[i][1]])
			parts.append("%s %s %s" % [_img(a, 20), ACTION_NAMES[a], " ".join(lv)])
		lines.append("  •  [b]%s[/b]: %s" % [c.name, "  ".join(parts)])
	return "\n".join(lines)

func refresh() -> void:
	if g == null or side == null:
		return
	_save()
	_update_board()
	while not g.volc_fx.is_empty():
		var fx: Dictionary = g.volc_fx.pop_front()
		board.lava(int(fx.from), int(fx.to), fx.dice)
	while not g.treb_fx.is_empty():
		var tf: Dictionary = g.treb_fx.pop_front()
		board.treb_anim.shot(int(tf.from), int(tf.to), int(tf.pid))
	_place_bubbles()
	_stats()
	_hex_info()
	for c in side.get_children():
		c.queue_free()
	_side_head()
	var body := VBoxContainer.new()
	body.size_flags_vertical = Control.SIZE_EXPAND_FILL
	body.add_theme_constant_override("separation", 8)
	side.add_child(body)
	if g.over >= 0:
		_game_over(body)
	elif busy or g.current != 0:
		_big_card(body, g.current, String(g.play.get("card", "")), "Playing", false)
	elif not g.volley.is_empty():
		if g.is_volc_hex(g.volley[0]):
			var vv: Dictionary = g.volc_levels()[g.volc_lv(0) - 1]
			_note_box(body, "[img=30]res://art/terrain/castle_dinos.png[/img] Your volcano is erupting: %d lava bombs for one enemy hex %s. Tap a hex ringed in red to aim it." % [int(vv.dice), "anywhere on the map" if int(vv.reach) >= 99 else "within %d hexes" % int(vv.reach)], GOLD)
		else:
			_note_box(body, "%s Your trebuchet is ready. Tap a hex ringed in red next to it to fire." % _img("advance", 22), GOLD)
		_battle_box(body)
	elif not info_unit.is_empty() and g.cur_action() == "recruit" and _sel_mine():
		_info_card(body)
	elif g.play.is_empty():
		if preview != "":
			_big_card(body, 0, preview, "Your card", true)
		else:
			_hand(body)
	else:
		_big_card(body, 0, String(g.play.card), "Playing", false)


func _update_board() -> void:
	board.selected = sel
	board.wall_for = wall_for if (g.cur_action() == "advance" and g.current == 0) else -1
	board.flash = ai_flash
	board.flash_color = Color(String(GameData.factions[g.players[g.current].faction].color))
	board.targets = []
	board.red_targets = false
	board.shooter = -1
	if g.over < 0 and g.current == 0 and not busy:
		if not g.volley.is_empty():
			board.shooter = g.volley[0]
			board.targets = g.shot_targets(g.volley[0])
			board.red_targets = true
		elif g.cur_action() == "move" and sel >= 0 and int(g.tiles[sel].owner) == 0 and int(g.tiles[sel].fresh) > 0 and g.moves_left() > 0:
			board.targets = g.move_targets(sel, g.cur_lvl())
		elif robo_up != "" and g.cur_action() == "advance":
			board.targets = g.robo_up_targets(0, robo_up)
		elif wall_for >= 0 and g.cur_action() == "advance":
			for j in g.wall_sides(wall_for):
				if g.wall_on(wall_for, j) < int(GameData.rules.walls.hp) and int(g.tiles[j].owner) != 0:
					board.targets.append(j)
	board.queue_redraw()

			mb.add_theme_stylebox_override(st, empty)

## A Dinosaur card's head shot (Ethan's OpenArt pictures), or null.
func _card_art(pid: int, cid: String, h: float) -> TextureRect:
	var path: String = String(GameData.factions[g.players[pid].faction].get("card_art", {}).get(cid, ""))
	if path == "" or not ResourceLoader.exists(path):
		return null
	var tx: Texture2D = load(path)
	var ar: float = clampf(tx.get_size().x / maxf(1.0, tx.get_size().y), 1.0, 1.45) if tx != null else 1.0   # the Robots' punch cards are wider
	var p := _pic(path, Vector2(h * ar, h))
	p.size_flags_vertical = Control.SIZE_SHRINK_CENTER
	p.mouse_filter = Control.MOUSE_FILTER_IGNORE
	return p


func _level_name(pid: int, a: String, l: int, cid: String = "") -> String:
	if g.d_up(pid, cid, a):   # the Dinosaurs: each card's own upgrades
		var d: Array = g.dcard(cid, a)
		return ", ".join(PackedStringArray(d.slice(0, l).map(func(x): return String(x[0]))))
	var named: Dictionary = GameData.upgrades[g.players[pid].faction + "_named"][a]
	if l == 1:
		return String(named.name)
	return "%s, %s" % [named.name, GameData.upgrades.general[a].now[l - 1]]

	return " " + "[img=16x16]res://art/ui/trophy.png[/img]".repeat(l)


## An action's icon: tap it for a box explaining the action (with its upgrades).

## What an action does at its current level, as in the browser version's actionDesc.
func _action_desc(a: String, pid: int, raw: int, cid: String = "") -> String:
	if g.is_myth(pid) and a != "fortify":
		return _myth_action_desc(a, pid)
	var lvl: int = 0 if (raw == 1 or g.card_ups(pid)) else raw   # the Dinosaurs' and Robots' cards use their own upgrades
	var m: Dictionary = g.mods_of(pid)
	var basic: String = g.basic_of(pid)
	var bname: String = _plural(basic)
	var txt: String = ""
	match a:
		"harvest":
			var base: int = int(m.harvestBase) if lvl == 0 else int(GameData.rules.actions.harvest.base_by_level[lvl])
			var per: float = float(GameData.rules.actions.harvest.per_hex_by_level[lvl])
			var per_txt: String = "1 for every 2 hexes" if per == 0.5 else ("1 for every hex" if per == 1.0 else "%d for every hex" % int(per))
			txt = "+%d gold, plus %s you own (on top of your normal income)" % [base, per_txt]
		"recruit":
			var per_troop: int = int(g.faction_of(pid).basic_per_troop)
			var cost: int = int(GameData.units[basic].cost)
			if lvl > 0:
				var fc: int = int(GameData.rules.actions.recruit.free_at_castle_by_level[lvl]) * per_troop
				var fb: int = int(GameData.rules.actions.recruit.free_at_barracks_by_level[lvl]) * per_troop
				txt = "First, free %s: %d at each castle and %d at each barracks. Then recruit %s (%d gold each) at your castles and barracks, or train other units at barracks%s" % [bname, fc, fb, bname, cost, "" if g.is_dinos(pid) else " and forts"]
			else:
				txt = "Recruit %s (%d gold each) at your castles and barracks, or train other units at barracks%s" % [bname, cost, "" if g.is_dinos(pid) else " and forts"]
		"build":
			var c: Dictionary = GameData.rules.buildings.cost_by_level
			if g.is_robots(pid):
				txt = "Build markets (%d gold). Stack up to %d on a hex for a resource hex, or, during Advancements, turn one into a factory at your home castle: a Barracks, then a Fortress (%s each). Buy materials at your markets" % [int(c.market[lvl]), int(GameData.rules.robots.markets_max), g.mat_txt(int(c.barracks[0]))]
			elif g.is_dinos(pid):
				txt = "Build as many as you can afford: Markets (%d gold) and Barracks (%s). Buy materials at your markets" % [int(c.market[lvl]), g.mat_txt(int(c.barracks[lvl]))]
			else:
				txt = "Build as many as you can afford: Markets (%d gold), Forts (%s) and Barracks (%s). Buy materials at your markets" % [int(c.market[lvl]), g.mat_txt(int(c.fort[lvl])), g.mat_txt(int(c.barracks[lvl]))]
		"move":
			var extra: Array = []
			if lvl >= 1:
				extra.append("troops can climb onto mountains")
			if lvl >= 2:
				extra.append("you attack with advantage")
			txt = "Make up to %d moves. A move sends troops %s to claim or attack, or up to %d hexes through your own land (not mountains)" % [int(m.movesPerAction), "up to 2 hexes" if lvl >= 3 else "to a neighboring hex", 3 if lvl >= 3 else 2]
			if not extra.is_empty():
				txt += ". With this card " + " and ".join(extra)
		"advance":
			var up_costs: Array = GameData.rules.buildings.barracks_upgrade_cost
			var tr: Dictionary = GameData.rules.trebuchet
			if g.is_dinos(pid):
				var D: Dictionary = GameData.rules.dinos
				txt = "Do as many as you can afford: upgrade a barracks (%d, %d, %d materials for levels 1 to 3%s), hire Alamosaurus guards for a barracks (%d gold each, up to %d per barracks), wake or stoke the volcano at your home castle (%d, %d, %d gold for levels 1 to 3), or build walls on a hex's sides (%d materials each)" % [int(up_costs[0]), int(up_costs[1]), int(up_costs[2]), ", your first level 1 free" if lvl >= 1 else "", int(D.guard_cost), int(D.guard_max), int(D.volcano[0].cost), int(D.volcano[1].cost), int(D.volcano[2].cost), int(GameData.rules.walls.cost_by_level[lvl])]
			else:
				txt = "Do as many as you can afford, all paid in materials: %supgrade a barracks (%d, %d, %d for levels 1 to 3%s), start a trebuchet on a hex with a fort (%d) or on a castle (up to 2: %d then %d), or build walls on a hex's sides (%d each)" % [("add a Barracks or Fortress to one of your markets at your home castle (%d each), " % int(GameData.rules.buildings.cost_by_level.barracks[0])) if g.is_robots(pid) else "", int(up_costs[0]), int(up_costs[1]), int(up_costs[2]), ", your first level 1 free" if lvl >= 1 else "", int(tr.field_cost), int(tr.castle_cost[0]), int(tr.castle_cost[1]), int(GameData.rules.walls.cost_by_level[lvl])]
		"fortify":
			var n: int = g.fx_picks_at(pid, raw)
			txt = "A wild card. First regroup: move your units up to %d hexes between your own hexes, as often as you like (no attacks or claiming). Then %s Build, Advancements and Recruit, played like the basic action" % [g.regroup_reach(), "do all three of" if n >= 3 else ("pick two of" if n == 2 else "pick one of")]
	if g.card_ups(pid):
		if raw >= 1 and g.d_up(pid, cid, a):
			var d: Array = g.dcard(cid, a)
			txt += ". " + ". ".join(PackedStringArray(d.slice(0, raw).map(func(x): return "%s: %s" % [x[0], x[1]])))
	elif raw >= 1:
		txt += ". " + String(GameData.upgrades[g.players[pid].faction + "_named"][a].desc)
	return txt


func _myth_action_desc(a: String, pid: int) -> String:
	var M: Dictionary = GameData.rules.mythical
	var joined := func(arr: Array) -> String: return ", ".join(PackedStringArray(arr.map(func(v): return str(int(v)))))
	match a:
		"harvest":
			return "+%d magic, plus 1 for every 2 hexes you own (on top of your Life Trees and castle)" % int(g.mods_of(pid).harvestBase)
		"recruit":
			return "At your castles: Minions (%d for 1 magic) and Mages (%d magic, at most %d on a hex). Where a Mage stands: turn Minions into Minotaurs, Griffins and Dragons" % [int(M.minion_pack), g.mage_price(pid), g.mage_cap(pid)]
		"build":
			return "Plant Life Trees: a Mage (even one that just moved) stays on its hex for %d turns; the tree makes %d magic a turn (%d with %d+ Minions there)" % [int(M.tree_turns), int(M.tree_magic), int(M.tree_rich), int(M.tree_minions)]
		"move":
			return "Make up to %d moves. Mages and Minions go up to 2 hexes anywhere; other units 1 hex into new land or an attack, or 2 through your own land. Only flyers can attack a Floating Castle" % g.moves_limit(pid)
		"advance":
			return "Upgrade your Mages (%s magic for levels 1 to 3; %s) or the Floating Castle (%s magic), or raise a magic dome where a Mage stands (%d magic, then %d a turn)" % [joined.call(M.mage_up), g.mage_req_all(), joined.call(M.castle_up), int(M.dome_cost), int(M.dome_upkeep)]
		"channel":
			return "No other actions this turn: +%d magic for every Mage you have and +%d for every castle you hold" % [int(M.ch_mage), int(M.ch_castle)]
	return ""


func _plural(k: String) -> String:
	var n: String = String(GameData.units[k].name)
	if n.ends_with("us") or n.ends_with("s"):
		return n
	return n + "s"

func _sel_mine() -> bool:
	return sel >= 0 and int(g.tiles[sel].owner) == 0


func _action_controls(box: VBoxContainer) -> void:
	var a0: String = g.cur_action()
	if g.fx_on():
		_fx_head(box)
		if a0 == "fortify":
			_fx_pick_controls(box)
			return
	if g.is_myth(0) and (a0 == "recruit" or a0 == "build" or a0 == "advance"):
		_myth_controls(box, a0)
		return
	match a0:
		"recruit":
			if g.dpick_waiting():
				box.add_child(_rich("[color=#efc15a]Calling: tap one of your nesting grounds for the free units: %s[/color]" % _army_icons(g.play.dpick), 14))
			_recruit_controls(box)
		"build":
			_mat_row(box)
			_build_controls(box)
		"advance":
			_mat_row(box)
			_advance_controls(box)
		"move":
			if g.regrouping():
				_regroup_controls(box)
			else:
				_move_controls(box)


## The heading while a Fortify runs: the regroup, or which pick is being played.
func _fx_head(box: VBoxContainer) -> void:
	var f: Dictionary = g.play.fx
	if f.ph == "regroup":
		box.add_child(_rich("%s [b]Regroup[/b]" % _img("fortify", 24), 16))
	elif f.ph == "act" and String(f.cur) == "move" and bool(f.get("mvx", false)):
		box.add_child(_rich("%s [b]Sortie: %s Move[/b]  [color=#a5b49e]a real Move, to claim or attack[/color]" % [_img("fortify", 24), _img("move", 24)], 16))
	elif f.ph == "act":
		box.add_child(_rich("%s [b]Fortify%s: %s %s[/b]" % [_img("fortify", 24), (" (%d of %d)" % [f.done.size() + 1, int(f.n)]) if int(f.n) > 1 else "", _img(String(f.cur), 24), ACTION_NAMES[String(f.cur)]], 16))


const FX_NOTES := {"build": "Markets, forts, barracks", "advance": "Barracks, trebuchets, walls", "recruit": "Train units", "move": "Claim and attack"}
const FX_NOTES_DINO := {"build": "Markets and barracks", "advance": "Barracks, guards, volcano, walls"}
const FX_NOTES_ROBO := {"build": "Markets", "advance": "Factory upgrades, barracks, trebuchets, walls"}

## Fortify's picks: Build, Advancements or Recruit (and Move with Pack Call's On the Move).
func _fx_pick_controls(box: VBoxContainer) -> void:
	var f: Dictionary = g.play.fx
	var left: int = int(f.n) - f.done.size()
	var head: String = "Pick one"
	if left <= 0:
		head = "Fortify done"
	elif int(f.n) >= 3:
		head = ("%d more to do, in any order" % left) if f.done.size() > 0 else "Do all three, in any order"
	elif f.done.size() > 0:
		head = "Pick one more"
	elif int(f.n) == 2:
		head = "Pick two, one at a time"
	box.add_child(_rich("%s [b]%s[/b]" % [_img("fortify", 24), head], 16))
	var grid := GridContainer.new()
	var picks: Array = g.fx_pick_list()
	grid.columns = picks.size()
	grid.add_theme_constant_override("h_separation", 6)
	box.add_child(grid)
	for k in picks:
		var did: bool = f.done.has(k)
		var note: String = String(FX_NOTES_MYTH.get(k, FX_NOTES[k])) if g.is_myth(0) else String((FX_NOTES_DINO if g.is_dinos(0) else (FX_NOTES_ROBO if g.is_robots(0) else {})).get(k, FX_NOTES[k]))
		var kk: String = k
		var b := _build_button("res://art/ui/%s.png" % k, ACTION_NAMES[k], "Done" if did else note, func(): _do(func(): g.fx_choose(kk)), not did and left > 0)
		b.size_flags_horizontal = Control.SIZE_EXPAND_FILL
		b.alignment = HORIZONTAL_ALIGNMENT_CENTER
		b.icon_alignment = HORIZONTAL_ALIGNMENT_CENTER
		b.vertical_icon_alignment = VERTICAL_ALIGNMENT_TOP
		b.add_theme_font_size_override("font_size", 13)
		b.autowrap_mode = TextServer.AUTOWRAP_WORD
		b.custom_minimum_size.y = 96
		grid.add_child(b)
	var tail: String = "Each plays like the basic action, without this card's upgrades."
	if g.uses_mat(0):
		tail += " You have %d gold and %s." % [int(g.players[0].gold), g.mat_txt(g.mat_of(0))]
	_hint(box, tail)

# =====================================================================
# Saving: the game is saved after everything you do; Continue on the title screen picks it up
# =====================================================================

func _save() -> void:
	if g == null or busy:
		return
	if g.over >= 0:
		DirAccess.remove_absolute(ProjectSettings.globalize_path(SAVE_FILE))
		return
	if g.current != 0 and not g.play.is_empty():
		return   # the computer is in the middle of its turn
	var f := FileAccess.open(SAVE_FILE, FileAccess.WRITE)
	if f != null:
		f.store_string(JSON.stringify({"v": 1, "game": g.snapshot()}))


func _read_save() -> Dictionary:
	if not FileAccess.file_exists(SAVE_FILE):
		return {}
	var txt: String = FileAccess.get_file_as_string(SAVE_FILE)
	var d = JSON.parse_string(txt)
	if not (d is Dictionary) or not d.has("game"):
		return {}
	d = GameState.ints(d)
	var gs: Dictionary = d.game
	if not gs.has("players") or gs.players.size() != 2 or int(gs.get("over", -1)) >= 0:
		return {}
	return d


func _continue_game(saved: Dictionary) -> void:
	my_faction = String(saved.game.players[0].faction)
	start_game(my_faction, saved.game)

func _ink(pid: int) -> String:

func _my_color_for(fid: String) -> String:
	return pref_color if COLORS.has(pref_color) else String(FAC_COLOR.get(fid, "blue"))


## The other side's colour: its faction's usual one, unless you took it.
func _other_color(taken, fid: String) -> String:
	var tk: Array = taken if taken is Array else [taken]
	var d: String = String(FAC_COLOR.get(fid, ""))
	if d != "" and not tk.has(d):
		return d
	for c in ["red", "blue", "purple", "green", "yellow"]:
		if not tk.has(c):
			return c
	return "red"


func _apply_colors() -> void:
	var mine: String = _my_color_for(String(g.players[0].faction))
	var taken: Array = [mine]
	var pcs: Array = [[0, mine]]
	for k in range(1, g.players.size()):
		var c: String = _other_color(taken, String(g.players[k].faction))
		taken.append(c)
		pcs.append([k, c])
	for pc in pcs:
		var C: Dictionary = COLORS[pc[1]]
		g.players[pc[0]]["col"] = pc[1]
		g.players[pc[0]]["color"] = C.color
		g.players[pc[0]]["ink"] = C.ink


func _set_pref_color(c: String) -> void:
	pref_color = c
	var cfg := ConfigFile.new()

## What you could still do in the current action, or "" if nothing useful is left.
func _leftover_warning() -> String:
	var a: String = g.cur_action()
	var gold: int = int(g.players[0].gold)
	var mats: int = g.mat_of(0)
	if g.fx_on():
		var f: Dictionary = g.play.fx
		if f.ph == "regroup":
			return ""   # regrouping is optional
		if f.ph == "pick":
			var left: int = int(f.n) - f.done.size()
			if left <= 0:
				return ""
			return "Your Fortify can still [b]%s[/b] of Build, Advancements and Recruit." % ("pick one" if (left == 1 and f.done.is_empty()) else ("pick one more" if left == 1 else "pick %d more" % left))
	if a == "recruit" and g.dpick_waiting():
		return "You haven't placed your free units from [b]Calling[/b] yet: tap one of your nesting grounds."
	if g.is_myth(0) and (a == "recruit" or a == "build" or a == "advance"):
		return _myth_leftover(a)
	var lvl: int = g.cur_lvl()
	var mine: Array = g.tiles.filter(func(t): return int(t.owner) == 0)
	match a:
		"move":
			var left: int = g.moves_left()
			if left <= 0:
				return ""
			var ready: int = 0
			for t in mine:
				if int(t.fresh) > 0 and not g.move_targets(int(t.id), lvl).is_empty():
					ready += int(t.fresh)
			if ready == 0:
				return ""
			return "You still have [b]%d move%s[/b] left, and [b]%d[/b] unit%s ready to go." % [left, "" if left == 1 else "s", ready, " is" if ready == 1 else "s are"]
		"recruit":
			for t in mine:
				if g.can_recruit_at(t) and gold >= g.unit_cost_at(int(t.id), 0, g.basic_of(0)):
					return "You have [b]%d gold[/b], enough to recruit or train more units." % gold
				for menu in ["barracks", "fort"]:
					if g.has_menu(t, menu):
						for k in g.faction_of(0).training[menu]:
							if g.train_ok(t, menu, k) and gold >= g.unit_cost_at(int(t.id), 0, k):
								return "You have [b]%d gold[/b], enough to recruit or train more units." % gold
		"build":
			for t in mine:
				if t.terrain == "plain" and g.building_count(t) < g.max_buildings() and (not g.is_robots(0) or g.robo_can_market(t)):
					for k in (["market"] if (t.castle or g.is_robots(0)) else (["market", "barracks"] if g.is_dinos(0) else ["market", "fort", "barracks"])):
						if g.can_pay_b(0, k, g.build_cost(k, t)):
							return "You have [b]%d gold[/b] and [b]%s[/b], and room to build more." % [gold, g.mat_txt(mats)]
		"advance":
			var adv_txt: String = "upgrade a barracks, hire a guard, stoke the volcano or build walls" if g.is_dinos(0) else "upgrade a barracks, start a trebuchet or build walls"
			if g.is_robots(0) and g.home_market(0):
				for k in ["barracks", "fort"]:
					if not g.robo_up_targets(0, k).is_empty() and mats >= g.robo_up_best(0, k):
						return "You have [b]%s[/b] and can still add a %s to one of your markets at your home castle." % [g.mat_txt(mats), "fortress" if k == "fort" else "barracks"]
			for t in mine:
				if t.terrain == "water":
					continue
				if g.is_dinos(0) and int(t.barracks) > 0 and g.guards_at(t) < g.guard_max(t) and gold >= g.guard_cost(0):
					return "You have [b]%d gold[/b] and can still %s." % [gold, adv_txt]
				if g.is_dinos(0) and g.is_home_castle(t) and g.volc_lv(0) < 3 and gold >= int(g.volc_levels()[g.volc_lv(0)].cost):
					return "You have [b]%d gold[/b] and can still %s." % [gold, adv_txt]
				if int(t.barracks) > 0 and int(t.bk) < 3:
					var free: bool = lvl >= 1 and not bool(g.play.free_bk) and int(t.bk) == 0
					if free or mats >= g.bk_up_cost(0, int(t.bk)):
						return "You have [b]%s[/b] and can still %s." % [g.mat_txt(mats), adv_txt]
				if g.can_treb_at(t) and mats >= g.treb_price(t):
					return "You have [b]%s[/b] and can still %s." % [g.mat_txt(mats), adv_txt]
				if mats >= g.wall_cost() or g.free_walls_left() > 0:
					for j in g.wall_sides(int(t.id)):
						if g.wall_on(int(t.id), j) < int(GameData.rules.walls.hp):
							return "You have [b]%s[/b] and can still %s." % [g.mat_txt(mats), adv_txt]
	return ""


func _myth_leftover(a: String) -> String:
	var magic: int = int(g.players[0].gold)
	for t in g.tiles:
		if int(t.owner) != 0:
			continue
		if a == "recruit" and ((t.castle and magic >= 1) or (g.mage_lv_at(t) >= 0 and int(t.army.get("minion", 0)) >= 1 and magic >= g.conv_price(t, "minotaur"))):
			return "You have [b]%d magic[/b], enough to make more units." % magic
		if a == "build" and int(t.tree) == 0 and not t.castle and t.terrain == "plain":
			if g.mages_on(t) > 0:
				return "A Mage of yours can still plant a [b]Life Tree[/b]."
	if a == "advance" and not g.myth_home(0).is_empty():
		var M: Dictionary = GameData.rules.mythical
		if (g.mlv_of(0) < 3 and magic >= int(M.mage_up[g.mlv_of(0)]) and g.mage_req_ok(0, g.mlv_of(0))) or (g.clv_of(0) < 3 and magic >= int(M.castle_up[g.clv_of(0)])):
			return "You have [b]%d magic[/b], enough for an upgrade." % magic
	return ""


func _confirm_leftover(msg: String) -> void:
	var last: bool = g.last_step_now()
	var box := _open_overlay("Still something to do", 480.0)
	box.add_child(_rich(msg, 16))
	box.add_child(_label("End your turn anyway?" if last else "Move on to the next action anyway? You can't come back to this one.", 14, LICHEN))
	var row := HBoxContainer.new()
	row.add_theme_constant_override("separation", 8)
	box.add_child(row)
	var keep := _button("Keep playing", func(): _close_overlay())
	keep.size_flags_horizontal = Control.SIZE_EXPAND_FILL
	row.add_child(keep)
	var go := _button("End turn" if last else "Next action", func():
		_close_overlay()
		_next_action(true), true)
	go.size_flags_horizontal = Control.SIZE_EXPAND_FILL
	row.add_child(go)
	keep.grab_focus()

## The Robots' home castle in Advancements: Barracks and Fortress (pick one, then tap the market to upgrade), and Overclock.
func _robo_home_bubbles(post: Callable, R: float) -> void:
	for spec in [["barracks", 1, 150], ["fort", 2, 210]]:
		var k: String = spec[0]
		var tg: Array[int] = g.robo_up_targets(0, k)
		var c: int = g.robo_up_best(0, k)
		var can: bool = not tg.is_empty() and g.mat_of(0) >= c
		var note: String = ("%d|mat" % c) if not tg.is_empty() else ("No barracks" if k == "fort" else "No market")
		_bubble(post.call(spec[1], spec[2]), R, "res://art/buildings/%s_robots.png" % k, "Fortress" if k == "fort" else "Barracks", note, can, robo_up == k, can,
			func():
				robo_up = k
				sel = -1
				refresh())
	if g.can_armor(0):   # Overclock (Construction, Advancements level 3)
		var ac: int = int(g.armor_rules().cost)
		var can_a: bool = not g.armor_on(0) and int(g.players[0].gold) >= ac
		_bubble(post.call(4, 330), R, "res://art/terrain/castle_robots.png", "Overclock", "Running" if g.armor_on(0) else "%d gold" % ac, can_a, false, can_a,
			func(): _do(func(): g.do_armor(0)))


func _place_bubbles() -> void:
	if bubbles == null:
		return
	for c in bubbles.get_children():
		c.queue_free()
	bubbles.posts = []
	bubbles.queue_redraw()
	if g == null or busy or g.current != 0 or g.over >= 0 or g.play.is_empty() or not g.volley.is_empty():
		return
	if not _sel_mine():
		return
	if bubbles.size.x < 10.0:   # the screen is still being laid out: place them once it has a size
		_place_bubbles.call_deferred()
		return
	var a: String = g.cur_action()
	var t: Dictionary = g.tiles[sel]
	var c: Vector2 = board.center(sel)
	var hr: float = board.radius()
	var R: float = clampf(hr * 0.66, 26.0, 44.0)
	var stem: float = R * 0.36
	if g.is_myth(0) and (a == "recruit" or a == "build" or a == "advance"):
		_myth_bubbles(t, c, hr, R, stem, a)
		bubbles.queue_redraw()
		return
	var fac: String = String(g.players[0].faction)
	var gold: int = int(g.players[0].gold)
	var lvl: int = g.cur_lvl()
	# a round button on a post, off corner vi of the hex, pointing out at angle deg
	var post := func(vi, deg) -> Vector2:
		var h: Vector2 = board._corner(c, hr, vi)
		var d := Vector2(cos(deg_to_rad(deg)), -sin(deg_to_rad(deg)))
		var at: Vector2 = h + d * (stem + R)
		at = at.clamp(Vector2(R + 2, R + 2), bubbles.size - Vector2(R + 2, R + 2))
		bubbles.posts.append([h, at - d * R * 0.9])
		return at
	if a == "build":
		if t.terrain != "plain":
			return
		var full: bool = g.building_count(t) >= g.max_buildings()
		for spot in [["barracks", 150, 1], ["fort", 90, 0], ["market", 30, 5]].filter(func(x): return not (x[0] == "fort" and g.is_dinos(0)) and (not g.is_robots(0) or x[0] == "market")):   # the Dinosaurs build no forts; the Robots build markets
			var kind: String = spot[0]
			var price: int = g.build_cost(kind, t)
			var castle_no: bool = t.castle and kind != "market"
			var can: bool = not full and not castle_no and g.can_pay_b(0, kind, price)
			var note: String = "Full" if full else ("Not here" if castle_no else ("Free" if price == 0 else (("%d|mat" % price) if g.in_mat(kind) else "%d gold" % price)))
			var title: String = kind.capitalize()
			if g.is_robots(0) and kind == "market" and not g.is_home_castle(t) and g.building_count(t) > 0:   # another market (a resource hex), or it's a factory
				var ok_m: bool = g.robo_can_market(t)
				can = ok_m and g.can_pay_b(0, "market", price)
				note = ("Free" if price == 0 else "%d gold" % price) if ok_m else ("Factory" if int(t.barracks) > 0 else "Full")
				title = ("Market %d" % (int(t.market) + 1)) if ok_m else "Market"
			var k2: String = kind
			_bubble(post.call(spot[2], spot[1]), R, "res://art/buildings/%s_%s.png" % [kind, fac], title, note, can, false, can,
				func(): _do(func(): g.build(sel, k2)))
		if g.mamm_now(0, "build"):
			var wp: int = g.build_cost("wh", t)
			var has_wh: bool = int(t.get("wh", 0)) > 0
			var can_wh: bool = not full and not has_wh and g.mat_of(0) >= wp
			_bubble(post.call(2, 210), R, "res://art/buildings/wh.png", "Watering Hole", "Built" if has_wh else ("Full" if full else "%d|mat" % wp), can_wh, false, can_wh,
				func(): _do(func(): g.build(sel, "wh")))
	elif a == "advance":
		if t.terrain == "water":
			return
		var lv: int = int(t.bk)
		var free: bool = lvl >= 1 and not bool(g.play.free_bk) and lv == 0
		var bp: int = 0 if (free or lv >= 3) else g.bk_up_cost(0, lv)
		var can_bk: bool = int(t.barracks) > 0 and lv < 3 and g.mat_of(0) >= bp
		var bk_note: String = "None here" if int(t.barracks) == 0 else ("Max level" if lv >= 3 else ("Free" if free else "%d|mat" % bp))
		if g.is_robots(0) and g.is_home_castle(t):   # the Robots' home castle: add a barracks or fortress to a market; Overclock
			_robo_home_bubbles(post, R)
		else:
			_bubble(post.call(1, 150), R, "res://art/buildings/barracks_%s.png" % fac, ("Level %d" % (lv + 1)) if (int(t.barracks) > 0 and lv < 3) else "Barracks", bk_note, can_bk, false, can_bk,
				func(): _do(func(): g.upgrade_barracks(sel)))
		if g.is_dinos(0):
			# the Dinosaurs: the volcano at the home castle, an Alamosaurus guard on a barracks
			if g.is_home_castle(t):
				var vl: int = g.volc_lv(0)
				var vc: int = int(g.volc_levels()[mini(2, vl)].cost)
				var can_v: bool = vl < 3 and gold >= vc
				_bubble(post.call(0, 90), R, "res://art/terrain/castle_dinos.png", ("Volcano %d" % (vl + 1)) if vl < 3 else "Volcano", "Max level" if vl >= 3 else "%d gold" % vc, can_v, false, can_v,
					func(): _do(func(): g.do_volcano(0)))
				if g.can_armor(0):   # Armor (The Hunt, Advancements level 3)
					var ac: int = int(g.armor_rules().cost)
					var can_a: bool = not g.armor_on(0) and gold >= ac
					_bubble(post.call(2, 210), R, "res://art/units/bow.png", "Armor", "Forged" if g.armor_on(0) else "%d gold" % ac, can_a, false, can_a,
						func(): _do(func(): g.do_armor(0)))
			else:
				var gn: int = g.guards_at(t)
				var gm: int = g.guard_max(t)
				var can_g: bool = int(t.barracks) > 0 and gn < gm and gold >= g.guard_cost(0)
				var g_note: String = "Needs barracks" if int(t.barracks) == 0 else (("%d guarding" % gn) if gn >= gm else "%d gold" % g.guard_cost(0))
				_bubble(post.call(0, 90), R, "res://art/units/alamo.png", "Alamosaurus", g_note, can_g, false, can_g,
					func(): _do(func(): g.buy_guard(sel)))
		else:
			var can_tr: bool = g.can_treb_at(t) and g.mat_of(0) >= g.treb_price(t)
			var tr_note: String = ("%d|mat" % g.treb_price(t)) if g.can_treb_at(t) else ("Max 2" if t.castle else ("Building" if int(t.treb) == 1 else ("Built" if int(t.treb) > 0 else "Needs fort")))
			_bubble(post.call(0, 90), R, "res://art/buildings/treb_%s.png" % fac, "Trebuchet 2" if (t.castle and g.treb_count(t) == 1) else "Trebuchet", tr_note, can_tr, false, can_tr,
				func(): _do(func(): g.start_trebuchet(sel)))
		var open_sides: bool = false
		for j in g.wall_sides(sel):
			if g.wall_on(sel, j) < int(GameData.rules.walls.hp):
				open_sides = true
		var wc: int = g.wall_cost()
		var can_w: bool = open_sides and (g.mat_of(0) >= wc or g.free_walls_left() > 0)
		var w_note: String = ("Free (%d left)" % g.free_walls_left() if g.free_walls_left() > 0 else "%d|mat" % wc) if open_sides else "All walled"
		_bubble(post.call(5, 30), R, "res://art/buildings/wall.png", "Walls", w_note, can_w, wall_for == sel, can_w,
			func():
				wall_for = -1 if wall_for == sel else sel
				refresh())
	elif a == "recruit":
		var basic: String = g.basic_of(0)
		if train_menu != "" and g.has_menu(t, train_menu):
			# the unit menu: the barracks' 4 units round the sides, or the fort's 3 along the top, and a hexagon below to go back
			var list: Array = g.faction_of(0).training[train_menu]
			var at: Array = [[1, 150], [5, 30], [2, 210], [4, 330]] if list.size() == 4 else [[1, 150], [0, 90], [5, 30], [2, 210], [4, 330]]
			var big: bool = list.size() > 5   # 6 units (the Dinosaurs) go evenly round the hex, with Back at the bottom
			var rr: float = minf(R, (hr * 0.95 + stem) / 1.75) if big else R
			var rring: float = hr * 0.95 + stem + rr
			var rpost := func(deg) -> Vector2:
				var d := Vector2(cos(deg_to_rad(deg)), -sin(deg_to_rad(deg)))
				var p0: Vector2 = (c + d * rring).clamp(Vector2(rr + 2, rr + 2), bubbles.size - Vector2(rr + 2, rr + 2))
				bubbles.posts.append([c + d * hr * 0.86, p0 - d * rr * 0.9])
				return p0
			var ring_at: Array = [167.1, 115.7, 64.3, 12.9, 321.4, 218.6]
			for n in (mini(list.size(), ring_at.size()) if big else mini(list.size(), at.size())):
				var k: String = list[n]
				var ok: bool = g.train_ok(t, train_menu, k)
				var note2: String = ("%d gold" % g.unit_cost_at(sel, 0, k)) if ok else "Level %d" % g.barracks_needed(k)
				var via: String = train_menu
				var picked: bool = info_unit.get("k", "") == k and info_unit.get("via", "") == via
				_bubble(rpost.call(float(ring_at[n])) if big else post.call(at[n][0], at[n][1]), rr, "res://art/units/%s.png" % k, _short(k), note2, ok, picked, true,
					func():
						info_unit = {"k": k, "via": via}
						refresh())
			_bubble(rpost.call(270.0) if big else post.call(3, 270), rr * 0.8, "", "Back", "", true, false, true, func():
				train_menu = ""
				info_unit = {}
				refresh())
		elif g.is_home_castle(t):
			# every unit round the home castle: the basic unit on top, barracks units down the left, fort units down the right.
			# Locked ones are faded and say which building unlocks them.
			var r: float = minf(R, (hr * 0.95 + stem) / 1.75)
			var ring: float = hr * 0.95 + stem + r
			var u: Dictionary = g.unlocks_of(0)
			var items: Array = [[basic, "basic", 90.0]]
			var bl: Array = g.faction_of(0).training.barracks.filter(func(x): return x != basic)
			var fl: Array = g.faction_of(0).training.fort
			for n in bl.size():
				items.append([bl[n], "barracks", float(UNIT_RING_BARRACKS[mini(n, 2)]) if not fl.is_empty() else float(UNIT_RING_ALL[mini(n, 4)])])
			for n in fl.size():
				items.append([fl[n], "fort", float(UNIT_RING_FORT[mini(n, 2)])])
			for it in items:
				var k: String = it[0]
				var via: String = it[1]
				var d := Vector2(cos(deg_to_rad(it[2])), -sin(deg_to_rad(it[2])))

		else:

## One round button: a picture, a name and a small note. A faded one ("off") still opens a unit's card,
## which says what unlocks it.
func _bubble(pos: Vector2, r: float, pic: String, title: String, note: String, on: bool, picked: bool, clickable: bool, f: Callable) -> void:
	var b := Button.new()
	b.position = pos - Vector2(r, r)
	b.size = Vector2(2 * r, 2 * r)
	b.custom_minimum_size = Vector2(2 * r, 2 * r)
	b.focus_mode = Control.FOCUS_NONE
	for st in ["normal", "hover", "pressed", "disabled"]:
		var sb := StyleBoxFlat.new()
		sb.bg_color = (Color("2a4430") if st == "hover" else MOSS2) if on else Color("1a231c")
		sb.set_corner_radius_all(int(r))
		sb.set_border_width_all(3 if picked else 2)
		sb.border_color = GOLD if picked else (OUTLINE if on else Color("2f3a31"))
		sb.shadow_color = Color(0, 0, 0, 0.45)
		sb.shadow_size = 3
		b.add_theme_stylebox_override(st, sb)
	b.disabled = not clickable
	if not on:
		b.modulate = Color(1, 1, 1, 0.62)
	var vb := VBoxContainer.new()
	vb.set_anchors_preset(Control.PRESET_FULL_RECT)
	vb.alignment = BoxContainer.ALIGNMENT_CENTER
	vb.add_theme_constant_override("separation", -2)
	b.add_child(vb)
	if pic != "":
		var p := _pic(pic, Vector2(r * 1.15, r * 0.72))
		p.size_flags_horizontal = Control.SIZE_SHRINK_CENTER
		vb.add_child(p)
	else:
		var hx := _label("⬢", int(r * 0.9), Color("2a5529"), true)
		hx.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
		vb.add_child(hx)
	var fs: int = clampi(int(r * 0.34), 10, 15)
	var tl := _label(title, fs, BONE, true)
	tl.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
	tl.clip_text = true
	tl.custom_minimum_size.x = 2 * r - 8
	vb.add_child(tl)

		var mrow := HBoxContainer.new()
		mrow.alignment = BoxContainer.ALIGNMENT_CENTER
		mrow.add_theme_constant_override("separation", 2)

		nl.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
		nl.clip_text = true
		nl.custom_minimum_size.x = 2 * r - 8
		vb.add_child(nl)
	_ignore_mouse(vb)
	b.pressed.connect(f)
	bubbles.add_child(b)

func _on_hex(i: int) -> void:
	if g == null or g.over >= 0:
		return
	if busy or g.current != 0:
		sel = -1 if sel == i else i
		_update_board()
		_hex_info()
		return
	if not g.volley.is_empty():
		if g.shot_targets(g.volley[0]).has(i):
			var before: int = g.log_n
			g.shoot_volley(g.volley[0], i)
			undo_stack = []   # the trebuchet (or volcano) rolls dice: its shot locks in
			_collect_notes(before)
		refresh()
		return
	if g.cur_action() == "recruit" and g.dpick_waiting() and int(g.tiles[i].owner) == 0 and int(g.tiles[i].barracks) > 0:
		sel = i   # Calling (Pack Call): the free units go to the nesting ground you tap
		_do(func(): g.place_dpick(i))
		return
	if g.cur_action() == "move" and sel >= 0 and board.targets.has(i):
		_open_move(sel, i)
		return
	if g.cur_action() == "advance" and robo_up != "":   # the Robots: tap a ringed market or factory to upgrade it
		if board.targets.has(i):
			var k0: String = robo_up
			_do(func(): g.do_robo_up(0, i, k0))
			if not g.robo_up_targets(0, k0).any(func(j): return g.mat_of(0) >= g.robo_up_cost(0, k0, g.tiles[j])):
				robo_up = ""   # nothing more to upgrade, or out of materials
				refresh()
			return
		robo_up = ""
	if g.cur_action() == "advance" and wall_for >= 0:
		if board.targets.has(i):
			var wf: int = wall_for
			_do(func(): g.build_wall(wf, i))
			return
		wall_for = -1
	sel = -1 if sel == i else i
	info_unit = {}
	train_menu = ""
	refresh()


## Does something you chose, saving the game first so Undo can put it back (only if something changed).
func _do(f: Callable) -> void:
	var before: Dictionary = g.snapshot()
	var was: String = JSON.stringify(before)
	f.call()
	if g.over < 0 and JSON.stringify(g.snapshot()) != was:
		undo_stack.append(before)
	refresh()


## Undo: takes back the last thing you did in this action (a move, a purchase, a building, an upgrade, a wall),
## as many steps back as you like, until you go on to the next action. Battles roll dice, so they lock in.
func _undo() -> void:
	if undo_stack.is_empty() or g.current != 0 or busy or g.play.is_empty():
		return
	g.restore(undo_stack.pop_back())
	if g.cur_action() == "move":
		sel = -1
	wall_for = -1
	robo_up = ""
	last_battle = {}
	refresh()


func _play(cid: String) -> void:
	if g.upgrade_due(0, cid):
		_open_upgrade(cid)
		return
	_start_card(cid, -1)


func _start_card(cid: String, slot: int) -> void:
	var before: int = g.log_n
	g.play_card(cid, slot)
	notes = []
	_collect_notes(before)
	preview = ""
	sel = -1
	refresh()


## Turns the log lines written by instant actions into the notes shown under your card.
func _collect_notes(before: int) -> void:
	var added: int = mini(g.log_n - before, g.log_lines.size())
	for k in range(added - 1, -1, -1):
		var line: String = g.log_lines[k]
		if line.begins_with("You play"):
			continue
		var icon: String = ""
		if "harvest" in line:
			icon = _img("harvest", 22)
		elif "fortify" in line or "Update" in line or "herd" in line:
			icon = _img("fortify", 22)
		elif "upgrade" in line:
			icon = "[color=#efc15a]▲[/color]"
		elif "free" in line or "factory" in line or "populate" in line:
			icon = _img("recruit", 22)
		elif "volcano" in line:
			icon = "[img=26]res://art/terrain/castle_dinos.png[/img]"
		elif "trebuchet" in line:
			icon = _img("advance", 22)
		elif "laser strike" in line:
			icon = _img("move", 22)
		notes.append("%s %s" % [icon, line])
	for n2 in g.step_notes:
		var a2: String = g.card_action()
		notes.append("%s %s" % [_img(a2 if a2 != "" else "fortify", 22), n2])
	g.step_notes.clear()


func _next_action(force: bool = false) -> void:
	if not force:
		var warn: String = _leftover_warning()
		if warn != "":
			_confirm_leftover(warn)
			return
	undo_stack = []
	info_unit = {}
	train_menu = ""
	sel = -1
	wall_for = -1
	robo_up = ""
	var before: int = g.log_n
	g.next_step()
	if g.current == 0:
		notes = []
		_collect_notes(before)
	if g.over < 0 and g.current != 0:
		notes = []
		ai_notes = []
		for k in range(mini(g.log_n - before, g.log_lines.size()) - 1, -1, -1):   # the computer's volcano, at the start of its turn
			if "volcano" in g.log_lines[k]:
				ai_notes.append("[img=26]res://art/terrain/castle_dinos.png[/img] " + g.log_lines[k])
		_save()
		_run_ai()
	refresh()


func _run_ai() -> void:
	busy = true
	ai_flash = []
	refresh()
	await get_tree().create_timer(0.8).timeout
	if g == null:
		return
	var game: GameState = g
	var pid: int = game.current
	var cid: String = AIPlayer.choose_card(game, pid)
	AIPlayer.start_card(game, pid, cid)
	refresh()
	var n: int = game.card_of(pid, cid).actions.size()
	for s in n:
		if game.over >= 0 or g != game:
			break
		game.play.step = s
		refresh()
		await get_tree().create_timer(2.5 + randf() * 1.0).timeout
		if g != game:
			return
		var events: Array = AIPlayer.run_step(game, pid, cid, s)
		_ai_events(events)
		refresh()
		await get_tree().create_timer(3.5 + randf() * 1.0).timeout
		if g != game:
			return
		# an attack on your units: you see it coming, then fight it out in the Arena; then the computer's moves go on
		while not events.is_empty() and events[events.size() - 1].kind == "fight" and game.over < 0:
			var fe: Dictionary = events[events.size() - 1]
			ai_flash = [int(fe.from), int(fe.to)]
			refresh()
			await get_tree().create_timer(0.9).timeout
			if g != game:
				return
			var r: Dictionary = await _fight(fe)
			if g != game:
				return
			ai_flash = [int(fe.from), int(fe.to)]
			refresh()
			await get_tree().create_timer(1.0).timeout
			if g != game or game.over >= 0:
				break
			events = AIPlayer.ai_moves(game, pid, int(fe.lvl))
			_ai_events(events)
			refresh()
			await get_tree().create_timer(1.4).timeout
			if g != game:
				return
	if game.over < 0:
		AIPlayer.ai_stock(game, pid)
		var before: int = game.log_n
		game.finish_turn()
		notes = []
		_collect_notes(before)
		var kept: Array = notes.filter(func(n2): return (String(n2).contains("trebuchet") or String(n2).contains("laser strike") or String(n2).contains("volcano") or String(n2).contains("out of the game")) and not String(n2).contains(String(game.players[pid].name) + " "))
		if game.current == 0:
			notes = ai_notes + kept
			ai_notes = []
		else:   # 3 players: the next computer goes on; its notes wait for your turn
			ai_notes = ai_notes + kept
			notes = []
	ai_flash = []
	busy = false
	refresh()
	if g == game and game.over < 0 and game.current != 0:   # 3 players: the next computer's turn
		_run_ai()


func _ai_events(events: Array) -> void:
	ai_flash = []
	for e in events:
		if e.kind == "done" or e.kind == "flash":
			ai_flash.append_array(e.ids)
		elif e.kind == "move":
			ai_flash.append(int(e.from))
			ai_flash.append(int(e.to))
			if e.res.get("kind", "") == "battle":
				last_battle = e.res


## Fights an attack in Megaraptor Arena over the whole screen, then applies the result to the board.
## The hex decides the map, and the defender holds it. Castle, fortress and mountain maps aren't built yet,
## so those battles are fought on the Open field for now.
func _fight(fight: Dictionary) -> Dictionary:
	var B: Dictionary = g.tiles[int(fight.to)]
	var map: String = g.arena_map(B)
	var built: Array = ["open", "ferns", "trees", "mountain", "castle", "fortress"]
	var map_names: Dictionary = {"open": "in the open", "castle": "for the castle", "fortress": "for the fortress", "mountain": "on the mountain", "trees": "in the forest", "ferns": "in the ferns"}
	var attacker: String = String(g.players[int(fight.att)].name)
	var defender: String = String(g.players[int(fight.def)].name)
	var sub: String = ("You attack the %s" % defender) if int(fight.att) == 0 else ("The %s attack you" % attacker)
	var play_map: String = map if built.has(map) else "open"
	if play_map != map:
		sub += "  ·  fought on open ground for now (the %s map comes in a later stage)" % map
	var fm: Dictionary = fight.get("mods", {})
	if bool(fm.get("sneak", false)):
		sub += "  ·  Sneak attack"
	if int(fm.get("plus", 0)) > 0:
		sub += "  ·  Updated"
	# the battle gets its own layer over the whole window
	var layer := CanvasLayer.new()
	layer.layer = 10
	add_child(layer)
	var screen = ArenaScreen.new()
	screen.theme = theme
	layer.add_child(screen)
	var fa: int = int(fight.att)
	var fd: int = int(fight.def)
	var rival: int = fd if fa == 0 else fa
	screen.start(g.fight_armies(fight), g.arena_team(0, rival), {"title": "Battle %s" % map_names.get(map, "in the open"), "sub": sub, "map": play_map, "hold": g.arena_team(fd, fa), "mods": g.arena_mods(fight), "cols": _arena_cols(fa, fd), "factions": _arena_factions(fa, fd), "ai_level": String(g.players[rival].get("diff", "medium"))}, _font_bold)
	var result: Dictionary = await screen.finished
	layer.queue_free()
	if g == null:
		return {}
	var r: Dictionary = g.finish_fight(fight, result)
	if r.get("kind", "") == "battle":
		last_battle = r
	return r

## Laser Strike: the tanks fire like a trebuchet. Dice are rolled, so it locks in (no undo).
func _laser(from: int, to: int) -> void:
	var r: Dictionary = g.do_laser(0, from, to)
	if r.is_empty():
		return
	undo_stack = []
	notes.append("%s Laser Strike: %d shot%s, %d hit%s, %d unit%s destroyed" % [_img("move", 22), int(r.shots), "" if int(r.shots) == 1 else "s", int(r.hits), "" if int(r.hits) == 1 else "s", int(r.killed), "" if int(r.killed) == 1 else "s"])
	sel = -1
	refresh()


func _go_move(from: int, to: int, units: Dictionary, join: bool = false) -> void:
	var before: int = g.log_n
	var snap: Dictionary = g.snapshot()
	var res: Dictionary = g.move(from, to, units, join)
	if res.get("kind", "") == "fight" or res.get("kind", "") == "battle":
		undo_stack = []   # battles roll dice, so they lock in
	elif not res.is_empty():
		undo_stack.append(snap)
	if res.get("kind", "") == "fight":   # your attack on their units: fought in Megaraptor Arena
		busy = true
		res = await _fight(res)
		busy = false
		if g == null:
			return
	if res.get("kind", "") == "battle":
		last_battle = res
	if res.has("coin"):
		_collect_notes(before)
		notes = notes.filter(func(n): return String(n).contains("coin"))
	sel = from if int(g.tiles[from].fresh) > 0 and g.moves_left() > 0 else -1
	refresh()

	if int(t.get("tree", 0)) > 0:
		bits.append("Life Tree, %d magic a turn" % g.tree_magic(t))
	if int(t.get("dome", 0)) > 0:
		bits.append("magic dome %d/%d" % [int(t.dome), int(GameData.rules.walls.hp)])
	if t.bounty:
		bits.append("bonus coin, %d gold" % int(GameData.rules.coin.gold))
	if not bits.is_empty():
		vb.add_child(_label(", ".join(bits), 13, LICHEN))
	if owner >= 0:
		vb.add_child(_label("Defense +%d%s" % [g.def_bonus(t), " · fortified" if g.braced(t) else ""], 13, LICHEN))


func _battle_box(parent: VBoxContainer) -> void:
	if last_battle.is_empty():
		return
	var r: Dictionary = last_battle
	var mine: bool = int(r.att) == 0
	var won: bool = bool(r.get("won", false))
	var head: String = ("You won the battle!" if won else "Your attack was pushed back.") if mine else ("The enemy broke through!" if won else "You held them off!")
	var good: bool = won == mine
	var p := PanelContainer.new()
	p.add_theme_stylebox_override("panel", _box(NIGHT, GOLD if good else Color("e0574c"), 2, 8))
	var vb := VBoxContainer.new()
	p.add_child(vb)
	vb.add_child(_rich("%s [b][color=%s]%s[/color][/b]" % ["[img=22x22]res://art/ui/swords.png[/img]", "#efc15a" if good else "#ff9a8f", head], 16))
	vb.add_child(_rich("[color=#a5b49e]Attackers lost:[/color] " + (_army_icons(r.get("att_lost", {})) if CombatRules.count(r.get("att_lost", {})) > 0 else "none"), 13))
	vb.add_child(_rich("[color=#a5b49e]Defenders lost:[/color] " + (_army_icons(r.get("def_lost", {})) if CombatRules.count(r.get("def_lost", {})) > 0 else "none"), 13))
	if int(r.get("wall_lost", 0)) > 0:
		vb.add_child(_label("Wall damage: %d" % int(r.wall_lost), 13, LICHEN))
	var dismiss := _button("OK", func():
		last_battle = {}
		refresh())
	dismiss.add_theme_stylebox_override("normal", _small_button_box())
	dismiss.size_flags_horizontal = Control.SIZE_SHRINK_END
	vb.add_child(dismiss)
	parent.add_child(p)

func _game_over(body: VBoxContainer) -> void:
	if g.players.size() > 2:
		_game_over3(body)
		return
	var won: bool = g.over == 0
	var why: Dictionary = {"home": "by taking the enemy home castle", "hold": "by holding %d castles" % int(GameData.rules.castles.to_win), "wipe": "by wiping the other side off the map"}
	var p := PanelContainer.new()

	vb.alignment = BoxContainer.ALIGNMENT_CENTER
	vb.add_theme_constant_override("separation", 10)
	p.add_child(vb)

	big.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
	big.add_theme_color_override("font_shadow_color", OUTLINE)
	big.add_theme_constant_override("shadow_offset_x", 3)
	big.add_theme_constant_override("shadow_offset_y", 3)

	vb.add_child(line)
	var again := _button("Play again", func(): show_factions(), true)
	again.size_flags_horizontal = Control.SIZE_SHRINK_CENTER
	vb.add_child(again)
	body.add_child(p)


## 3 players: the last side standing, or you were knocked out.
func _game_over3(body: VBoxContainer) -> void:
	var won: bool = g.over == 0
	var me: Dictionary = g.players[0]
	var p := PanelContainer.new()
	p.add_theme_stylebox_override("panel", _frame(MOSS2, 16))
	p.size_flags_vertical = Control.SIZE_EXPAND_FILL
	var vb := VBoxContainer.new()
	vb.alignment = BoxContainer.ALIGNMENT_CENTER
	vb.add_theme_constant_override("separation", 10)
	p.add_child(vb)
	var big := _label("You rule the land!" if won else ("You are out" if g.is_out(0) else "The %s rule the land" % g.players[g.over].name), 36, GOLD if won else RED, true)
	big.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
	big.autowrap_mode = TextServer.AUTOWRAP_WORD_SMART
	big.add_theme_color_override("font_shadow_color", OUTLINE)
	big.add_theme_constant_override("shadow_offset_x", 3)
	big.add_theme_constant_override("shadow_offset_y", 3)
	vb.add_child(big)
	var txt: String = ""
	if won:
		var parts: PackedStringArray = []
		for q in g.players:
			if int(q.id) == 0:
				continue
			var by: int = int(q.get("out_by", -1))
			parts.append("The %s fell in round %d%s." % [q.name, int(q.get("out_round", g.turn)), " to you" if by == 0 else (" to the %s" % g.players[by].name if by > 0 else "")])
		txt = "You're the last side standing. " + " ".join(parts)
	else:
		var by2: int = int(me.get("out_by", -1))
		var why: String = String(me.get("out_why", ""))
		txt = ("The %s took your home castle" % g.players[by2].name) if (why == "home" and by2 >= 0) else ("You have no land left" if why == "wipe" else "You were knocked out")
		txt += " in round %d." % int(me.get("out_round", g.turn))
		if g.over_why == "last":
			txt += " The %s were the last side standing." % g.players[g.over].name
	var line := _label(txt, 17, BONE)
	line.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
	line.autowrap_mode = TextServer.AUTOWRAP_WORD_SMART
	vb.add_child(line)
	var again := _button("Play again" if won else "Try again", func(): show_factions(), true)
	again.size_flags_horizontal = Control.SIZE_SHRINK_CENTER
	vb.add_child(again)
	body.add_child(p)

# Small builders

func _label(text: String, size_px: int, col: Color, bold: bool = false) -> Label:
	var l := Label.new()
	l.text = text
	# Short labels (names, numbers, headings) never wrap, so they can't be squeezed into one letter
	# per line. Sentences wrap and take the full width they're given.
	if text.length() > 34 or text.contains("\n"):
		l.autowrap_mode = TextServer.AUTOWRAP_WORD_SMART
		l.size_flags_horizontal = Control.SIZE_EXPAND_FILL
		l.custom_minimum_size.x = maxf(l.custom_minimum_size.x, 120.0)
	else:
		l.autowrap_mode = TextServer.AUTOWRAP_OFF
	l.add_theme_font_size_override("font_size", size_px)
	l.add_theme_color_override("font_color", col)
	if bold:
		l.add_theme_font_override("font", _font_bold)
	return l


func _rich(bbcode: String, size_px: int) -> RichTextLabel:
	var r := RichTextLabel.new()
	r.bbcode_enabled = true
	r.fit_content = true
	r.scroll_active = false
	r.autowrap_mode = TextServer.AUTOWRAP_WORD_SMART
	r.size_flags_horizontal = Control.SIZE_EXPAND_FILL
	r.mouse_filter = Control.MOUSE_FILTER_IGNORE
	r.add_theme_font_size_override("normal_font_size", size_px)
	r.add_theme_font_size_override("bold_font_size", size_px)
	r.text = bbcode
	return r


## An action icon inline in rich text.
func _img(action: String, size_px: int) -> String:
	return "[img=%dx%d]res://art/ui/%s.png[/img]" % [size_px, size_px, action]


func _pic(path: String, box: Vector2) -> TextureRect:
	var tr := TextureRect.new()
	if ResourceLoader.exists(path):
		tr.texture = load(path)
	tr.expand_mode = TextureRect.EXPAND_IGNORE_SIZE
	tr.stretch_mode = TextureRect.STRETCH_KEEP_ASPECT_CENTERED
	tr.custom_minimum_size = box
	tr.mouse_filter = Control.MOUSE_FILTER_IGNORE
	return tr


func _button(text: String, f: Callable, primary: bool = false) -> Button:
	var b := Button.new()
	b.text = text
	if primary:
		b.theme_type_variation = "Primary"
	b.pressed.connect(f)
	return b


func _ignore_mouse(n: Node) -> void:
	if n is Control:
		n.mouse_filter = Control.MOUSE_FILTER_IGNORE
	for c in n.get_children():
		_ignore_mouse(c)
